FIX A — "it still asks me to assign the file after ingest"
------------------------------------------------------------------------
The attach happens in _ingest_worker (line ~359) and needs BOTH keys:

    if category_key and product_key:
        catalog_mod.attach_source(category_key, product_key, filename)

So it silently skips whenever either header arrives as None. The usual
cause is header-name mismatch: FastAPI's Header() converts underscores to
hyphens by default, so a parameter named `category_key` expects the HTTP
header `category-key`. If the admin UI sends `category_key`, FastAPI sees
nothing and passes None — no error, no attach, and the document then shows
up in the "needs assignment" list exactly as you're seeing.

Make /upload accept either spelling, and log loudly when neither arrives:

    from fastapi import Request

    @app.post("/upload")
    async def upload(request: Request,
                     file: UploadFile = File(...),
                     x_user_id: str | None = Header(default=None)):
        # v3.4.0: read the scope headers off the raw request so BOTH
        # "category-key" and "category_key" work. Previously these were
        # Header() params, which only matched the hyphenated form — a
        # mismatch silently produced an unassigned document.
        _h = request.headers
        category_key = _h.get("category-key") or _h.get("category_key")
        product_key = _h.get("product-key") or _h.get("product_key")

        if not (category_key and product_key):
            logger.warning(
                f"upload: missing scope headers (category={category_key!r}, "
                f"product={product_key!r}) - document will need manual "
                f"assignment. Headers seen: {list(_h.keys())}")
        ...
        # rest unchanged; pass category_key / product_key to the worker
        # as before, and DROP the ingest_provider argument if you have
        # applied the doc2query removal.

That `Headers seen:` line is the diagnostic — one upload attempt tells you
exactly what the frontend is sending.

Also worth checking while you're in here: /admin/reassign_source writes

    m["product"] = payload.product_key          # line ~805

but ingest.py writes the metadata key "products" (plural). If retrieval
and FAQ scoping read one and reassign writes the other, a reassigned
document can end up scoped differently from an ingested one. Confirm which
key each reader expects and make all three agree.

FIX B — /faq/generate must MERGE, not replace
------------------------------------------------------------------------
Replace the body of faq_generate (line ~746). record_questions() wipes
every entry for a source, so regenerating destroys manual additions and
resets curated answers. merge_questions() adds only genuinely new
questions and never overwrites an existing one.

    class FaqGenerateReq(BaseModel):
        source: str
        product: str
        category: str = ""
        questions: list[dict]
        # v3.4.0: default is now non-destructive. Pass replace=true only
        # for a deliberate "regenerate this document from scratch".
        replace: bool = False

    @app.post("/faq/generate")
    def faq_generate(payload: FaqGenerateReq,
                     x_admin_password: str | None = Header(default=None)):
        """Store FAQ questions generated by the frontend.

        v3.4.0: merges by default. Questions already present in the same
        product scope are skipped (compared case- and punctuation-
        insensitively), and existing answers are never touched — so
        pressing Generate twice is safe and additive.
        """
        _require_admin(x_admin_password)
        qa = [{"question": q.get("question", "").strip(),
               "answer": q.get("answer", "")}
              for q in payload.questions if q.get("question", "").strip()]
        if not qa:
            raise HTTPException(status_code=400, detail="No questions provided")

        if payload.replace:
            faq_store.record_questions(payload.source, payload.product, qa,
                                       category=payload.category)
            return {"stored": len(qa), "replaced": True, "source": payload.source}

        result = faq_store.merge_questions(payload.source, payload.product, qa,
                                           category=payload.category)
        return {**result, "replaced": False, "source": payload.source}

NEW — manual add, edit question text, bulk delete
------------------------------------------------------------------------

    class FaqCreate(BaseModel):
        question: str
        answer: str = ""
        product: str = ""
        category: str = ""

    @app.post("/faq")
    def faq_create(payload: FaqCreate,
                   x_admin_password: str | None = Header(default=None)):
        """Add a question by hand. Marked origin="manual" and edited=True,
        so a later Generate can never overwrite or duplicate it."""
        _require_admin(x_admin_password)
        try:
            return faq_store.add_entry(payload.question, payload.answer,
                                       products=payload.product,
                                       category=payload.category)
        except ValueError as e:
            # 409 so the UI can show "that question already exists"
            # rather than a generic failure.
            raise HTTPException(status_code=409, detail=str(e))


    # REPLACE the existing FaqEdit model + PATCH handler so the QUESTION
    # can be edited too, not just the answer.
    class FaqEdit(BaseModel):
        question: str | None = None
        answer: str | None = None

    @app.patch("/faq/{faq_id}")
    def faq_edit(faq_id: str, payload: FaqEdit,
                 x_admin_password: str | None = Header(default=None)):
        _require_admin(x_admin_password)
        updated = faq_store.update_entry(faq_id, payload.question, payload.answer)
        if not updated:
            raise HTTPException(status_code=404, detail="FAQ entry not found")
        return updated


    @app.delete("/faq")
    def faq_delete_all(product: str | None = None,
                       source: str | None = None,
                       confirm: bool = False,
                       x_admin_password: str | None = Header(default=None)):
        """Bulk delete FAQ entries.

        Requires ?confirm=true. That's a deliberate second lock behind the
        UI's warning dialog: a mis-fired DELETE with no scope wipes the
        entire curated FAQ, which is hand-written content with no undo.
        """
        _require_admin(x_admin_password)
        if not confirm:
            raise HTTPException(
                status_code=400,
                detail="Refusing bulk delete without confirm=true")
        removed = faq_store.delete_all(scope_key=product, source=source)
        return {"deleted": removed,
                "scope": product or source or "ALL"}
